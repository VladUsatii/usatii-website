Live app lookup benchmark — 2026-10-09

Use signed-in Chrome tabs on corresponding Work Flow boards. Bind their CUA raw tab handles as wfDom and mondayDom. Choose three shared unique full task titles; keep customer/task text out of published observations. Set lookupTasks to those titles and appTrials to an empty array. Warm both boards before measuring. Execute paired batches in order U/M, M/U, U/M, M/U, U/M. Each batch repeats the three queries three times. Browser automation overhead is included; this is not API or database timing.

Executed routine:
async function runLookupBatch(product,repeat){let tab=product==='USATII'?wfDom:mondayDom;let input=product==='USATII'?tab.playwright.getByPlaceholder('Search',{exact:true}):tab.playwright.getByRole('textbox',{name:'Search / Filter Board'});let cumulative=0;for(let i=0;i<9;i++){let query=lookupTasks[i%3];let start=Date.now();await input.fill(query);let snap='',counts=[],ready=false,polls=0;while(Date.now()-start<15000){snap=await tab.playwright.domSnapshot();polls++;counts=[...snap.matchAll(product==='USATII'?/generic: (\d+) tasks?\b/g:/generic: (\d+) Tasks?\b/g)].map(x=>Number(x[1]));ready=counts.reduce((a,b)=>a+b,0)===1 && snap.includes(product==='USATII'?'textbox: '+query:'heading "'+query+'" [level=4]');if(ready)break;}let ms=Date.now()-start;cumulative+=ms;appTrials.push({product,repeat,lookup:i+1,taskId:i%3+1,ms,cumulativeMs:cumulative,matched:counts.reduce((a,b)=>a+b,0),ready,polls});if(!ready)throw Error('Lookup failed '+product+' '+i); }nodeRepl.write({product,repeat,totalMs:cumulative});}
